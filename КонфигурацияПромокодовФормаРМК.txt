Процедура Расш1_ПромокодПослеВвода(Результат, ДополнительныеПараметры) Экспорт

    Если Результат = Неопределено Тогда
        Возврат;
    КонецЕсли;

    КодПромокода = ВРег(СокрЛП(Результат));

    Если СтрНачинаетсяС(КодПромокода, "PROMO:") Тогда
        КодПромокода = СокрЛП(Сред(КодПромокода, 7));
    КонецЕсли;

    Если ПустаяСтрока(КодПромокода) Тогда
        Возврат;
    КонецЕсли;

    Если СтрДлина(КодПромокода) > 10 Тогда
        ПоказатьПредупреждение(, "Некорректная длина промокода.");
        Возврат;
    КонецЕсли;

    Если ЗначениеЗаполнено(Расш1_ПромокодВТекущемЧеке) Тогда
        ПоказатьПредупреждение(, "В этом чеке уже применен промокод: "
            + Расш1_ПромокодВТекущемЧеке);
        Возврат;
    КонецЕсли;

    Если НЕ ОбщегоНазначенияРМККлиент.ЭтоВидОперацииПродажа(Объект.ВидОперации)
        Или ЭтоЧекКоррекции
        Или НЕ (РежимРаботыСЧеком = РежимыРаботыСЧеком.ВводТоваров) Тогда
        ПоказатьПредупреждение(, "Промокод можно применить только при вводе товаров в чек продажи.");
        Возврат;
    КонецЕсли;

    Если Объект.Товары.Количество() = 0 Тогда
        ПоказатьПредупреждение(, "Сначала добавьте товары в чек.");
        Возврат;
    КонецЕсли;

    ДанныеПромокода = Расш1_ПолучитьДанныеПромокода(КодПромокода);

    Если НЕ ДанныеПромокода.Найден Тогда
        ПоказатьПредупреждение(, ДанныеПромокода.ТекстОшибки);
        Возврат;
    КонецЕсли;

    Если ЗначениеЗаполнено(Расш1_КодРезерва) Тогда
        ПоказатьПредупреждение(, "В чеке уже есть резерв промокода. Завершите или отмените чек.");
        Возврат;
    КонецЕсли;

    // Резервируем до изменения корзины. При неопределенном ответе
    // сохраняем идентификатор и запрещаем предоставление скидки.
    Расш1_КодРезерва = КодПромокода;
    Расш1_ТокенРезерва = Строка(Новый УникальныйИдентификатор);
    Расш1_НачатаФискализация = Ложь;
    Попытка
        ОтветAPI = Расш1_ЗапросПромоAPI("reserve", КодПромокода, Расш1_ТокенРезерва, ДанныеПромокода.ПараметрыПромоAPI);
        Если НЕ ОтветAPI.ok Тогда
            // При явном отказе 409/404 резерв не получен.
            Если ОтветAPI.error = "reserved" Или ОтветAPI.error = "used"
                Или ОтветAPI.error = "not_found" Или ОтветAPI.error = "code"
                Или ОтветAPI.error = "auth" Или ОтветAPI.error = "disabled"
                Или ОтветAPI.error = "metadata_mismatch" Или ОтветAPI.error = "expired"
                Или ОтветAPI.error = "metadata" Или ОтветAPI.error = "date"
                Или ОтветAPI.error = "gift" Или ОтветAPI.error = "discount"
                Или ОтветAPI.error = "type" Тогда
                Расш1_КодРезерва = "";
                Расш1_ТокенРезерва = "";
            КонецЕсли;
            ПоказатьПредупреждение(, ОтветAPI.message);
            Возврат;
        КонецЕсли;
        Если ОтветAPI.status <> "reserved" Тогда
            ВызватьИсключение "Неожиданный статус резерва.";
        КонецЕсли;
    Исключение
        ПоказатьПредупреждение(, "Не удалось подтвердить резерв промокода. Скидка не предоставлена. "
            + ОписаниеОшибки());
        Возврат;
    КонецПопытки;

    Если ДанныеПромокода.ТипПромокода =
        ПредопределенноеЗначение("Перечисление.ТипыПромокодов.Скидка") Тогда

        Если Объект.Товары.Количество() = 0 Тогда
            ПоказатьПредупреждение(, "Сначала добавьте товары в чек.");
            Возврат;
        КонецЕсли;

        Если ДанныеПромокода.ПроцентСкидки <= 0
            Или ДанныеПромокода.ПроцентСкидки > 100 Тогда
            ПоказатьПредупреждение(, "Некорректный процент скидки промокода.");
            Возврат;
        КонецЕсли;

        Если ПримененаСкидкаНаЧек
            Или Объект.Товары.Итог("СуммаСкидкиНаценки") <> 0 Тогда
            ПоказатьПредупреждение(, "Сначала отмените ручную скидку в чеке.");
            Возврат;
        КонецЕсли;

        // Разрешение действует только внутри расчета проверенного промокода.
        ИсходныйПределСкидки = МаксимальныйПроцентРучнойСкидки;
        Попытка
            МаксимальныйПроцентРучнойСкидки = Макс(
                ИсходныйПределСкидки, ДанныеПромокода.ПроцентСкидки);

            РассчитатьРучнуюСкидкуПроцентомНаПозиции(
                Объект.Товары,
                ДанныеПромокода.ПроцентСкидки);
        Исключение
            МаксимальныйПроцентРучнойСкидки = ИсходныйПределСкидки;
            ВызватьИсключение;
        КонецПопытки;
        МаксимальныйПроцентРучнойСкидки = ИсходныйПределСкидки;

        Если Объект.Товары.Итог("СуммаСкидкиНаценки") <= 0 Тогда
            ПоказатьПредупреждение(, "Промокод не применен: скидка на товары чека недоступна.");
            Возврат;
        КонецЕсли;

        ПересчитатьИтогиТабличнойЧастиТовары(Ложь);
        РассчитатьСуммыОплаты();

    ИначеЕсли ДанныеПромокода.ТипПромокода =
        ПредопределенноеЗначение("Перечисление.ТипыПромокодов.Подарок") Тогда

		СтруктураПодарка = ПодготовитьДанныеВыбораНоменклатуры(
		    ДанныеПромокода.Номенклатура);

		СтруктураПодарка.Вставить(
		    "Количество",
		    ДанныеПромокода.Количество);

		СтруктураПодарка.Вставить(
		    "КоличествоУпаковок",
		    ДанныеПромокода.Количество);

		СтруктураПодарка.Вставить(
		    "ОбластьПредоставленияДокумент",
		    Ложь);

		СтруктураПодарка.Вставить(
		    "ВыводитьПодарокВЧек",
		    Истина);
			
        КоличествоСтрокДо = Объект.Товары.Количество();
        ОбработатьВыдачуПодарка(СтруктураПодарка);

        Если Объект.Товары.Количество() <= КоличествоСтрокДо Тогда
            ПоказатьПредупреждение(, "Не удалось добавить подарок.");
            Возврат;
        КонецЕсли;

        СтрокаПодарка = Объект.Товары[Объект.Товары.Количество() - 1];
        Если НЕ СтрокаПодарка.Подарок
            Или СтрокаПодарка.Номенклатура <> ДанныеПромокода.Номенклатура Тогда
            ПоказатьПредупреждение(, "Не найдена строка подарка.");
            Возврат;
        КонецЕсли;

        Расш1_ИдентификаторПодарка = Строка(СтрокаПодарка.ИдентификаторСтроки);

    Иначе
        ПоказатьПредупреждение(, "Неизвестный тип промокода.");
        Возврат;
    КонецЕсли;

    Расш1_ПромокодВТекущемЧеке = КодПромокода;
    Расш1_ОбновитьКнопкуПромокода();

Функция Расш1_ПолучитьДанныеПромокода(КодПромокода)

    Результат = Новый Структура;
    Результат.Вставить("Найден", Ложь);
    Результат.Вставить("ТекстОшибки", "");
    Результат.Вставить("ТипПромокода", Неопределено);
    Результат.Вставить("Номенклатура", Неопределено);
    Результат.Вставить("Количество", 0);
    Результат.Вставить("ПроцентСкидки", 0);

    Запрос = Новый Запрос;
    Запрос.Текст =
    "ВЫБРАТЬ ПЕРВЫЕ 1
    |   Промокоды.ТипПромокода КАК ТипПромокода,
    |   Промокоды.Номенклатура КАК Номенклатура,
    |   Промокоды.Количество КАК Количество,
    |   Промокоды.ПроцентСкидки КАК ПроцентСкидки,
    |   Промокоды.ДатаНачала КАК ДатаНачала,
    |   Промокоды.ДатаОкончания КАК ДатаОкончания,
    |   Промокоды.Статус КАК Статус
    |ИЗ
    |   Справочник.Промокоды КАК Промокоды
    |ГДЕ
    |   Промокоды.Промокод = &Промокод
    |   И НЕ Промокоды.ПометкаУдаления";

    Запрос.УстановитьПараметр("Промокод", КодПромокода);

    Выборка = Запрос.Выполнить().Выбрать();

    Если НЕ Выборка.Следующий() Тогда
        Результат.ТекстОшибки = "Такой промокод не найден.";
        Возврат Результат;
    КонецЕсли;

    Если Выборка.Статус <>
        ПредопределенноеЗначение("Перечисление.СтатусыПромокодов.Активно") Тогда

        Результат.ТекстОшибки = "Этот промокод уже использован или неактивен.";
        Возврат Результат;

    КонецЕсли;

    Если ТекущаяДата() < НачалоДня(Выборка.ДатаНачала)
        ИЛИ ТекущаяДата() > КонецДня(Выборка.ДатаОкончания) Тогда

        Результат.ТекстОшибки = "Срок действия промокода истек или еще не наступил.";
        Возврат Результат;

    КонецЕсли;

    Результат.Найден = Истина;
    Результат.ТипПромокода = Выборка.ТипПромокода;
    Результат.Номенклатура = Выборка.Номенклатура;
    Результат.Количество = Выборка.Количество;
    Результат.ПроцентСкидки = Выборка.ПроцентСкидки;
    ПараметрыПромоAPI = Новый Структура;
    ПараметрыПромоAPI.Вставить("valid_from", Формат(Выборка.ДатаНачала, "ДФ=yyyy-MM-dd"));
    ПараметрыПромоAPI.Вставить("valid_until", Формат(Выборка.ДатаОкончания, "ДФ=yyyy-MM-dd"));
    Если Выборка.ТипПромокода = Перечисления.ТипыПромокодов.Подарок Тогда
        ПараметрыПромоAPI.Вставить("promo_type", "gift");
        ПараметрыПромоAPI.Вставить("product_sku", СокрЛП(Выборка.Номенклатура.Код));
        ПараметрыПромоAPI.Вставить("gift_quantity", Выборка.Количество);
    ИначеЕсли Выборка.ТипПромокода = Перечисления.ТипыПромокодов.Скидка Тогда
        ПараметрыПромоAPI.Вставить("promo_type", "discount");
        ПараметрыПромоAPI.Вставить("discount_percent", Выборка.ПроцентСкидки);
    Иначе
        Результат.Найден = Ложь;
        Результат.ТекстОшибки = "Неизвестный тип промокода.";
        Возврат Результат;
    КонецЕсли;
    Результат.Вставить("ПараметрыПромоAPI", ПараметрыПромоAPI);

    Возврат Результат;

Процедура Расш1_ПогаситьПромокодНаСервере(КодПромокода)

    НачатьТранзакцию();
    Попытка
        Запрос = Новый Запрос;
        Запрос.Текст =
        "ВЫБРАТЬ ПЕРВЫЕ 1
        |   Промокоды.Ссылка КАК Ссылка
        |ИЗ
        |   Справочник.Промокоды КАК Промокоды
        |ГДЕ
        |   Промокоды.Промокод = &Промокод
        |   И НЕ Промокоды.ПометкаУдаления";
        Запрос.УстановитьПараметр("Промокод", КодПромокода);
        Выборка = Запрос.Выполнить().Выбрать();
        Если НЕ Выборка.Следующий() Тогда
            ВызватьИсключение "Промокод не найден.";
        КонецЕсли;

        Блокировка = Новый БлокировкаДанных;
        ЭлементБлокировки = Блокировка.Добавить("Справочник.Промокоды");
        ЭлементБлокировки.Режим = РежимБлокировкиДанных.Исключительный;
        ЭлементБлокировки.УстановитьЗначение("Ссылка", Выборка.Ссылка);
        Блокировка.Заблокировать();

        ПромокодОбъект = Выборка.Ссылка.ПолучитьОбъект();
        Если ПромокодОбъект.Статус <> Перечисления.СтатусыПромокодов.Активно Тогда
            ВызватьИсключение "Промокод уже использован или неактивен.";
        КонецЕсли;
        ПромокодОбъект.Статус = Перечисления.СтатусыПромокодов.Применен;
        ПромокодОбъект.Записать();
        ЗафиксироватьТранзакцию();
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;

    Расш1_ОтменаПромокодаОжидаетОтвета = Ложь;
    Элементы.ПрименитьПромокодNew.Заголовок = "Промокод";
    Элементы.ПрименитьПромокодNew.ЦветТекста = ЦветаФормы.Авто;
КонецПроцедуры




// QR-коды нашего справочника имеют отдельный префикс PROMO:.
// Все остальные штрихкоды передаются штатной процедуре без изменений.
&НаКлиенте
&Вместо("ВыполнитьПоискШтрихкодаПродолжение")
Процедура Расш1_ОбработатьQRПромокода(Штрихкод)

    Если ТипЗнч(Штрихкод) <> Тип("Строка") Тогда
        ПродолжитьВызов(Штрихкод);
        Возврат;
    КонецЕсли;

    КодQR = ВРег(СокрЛП(Штрихкод));
    Если НЕ СтрНачинаетсяС(КодQR, "PROMO:") Тогда
        ПродолжитьВызов(Штрихкод);
        Возврат;
    КонецЕсли;

    Если НЕ (РежимРаботыСЧеком = РежимыРаботыСЧеком.ВводТоваров)
        Или ОжидатьВводМарки
        Или ИнтерфейсРМКСлужебныйКлиентСервер.ПолучитьТекущийСлойГруппы(
            Элементы.ГруппаСлоевСлужебнаяИнформацияЧека)
            <> Элементы.ОбластьОперативногоУправленияЧеком Тогда

        ПоказатьПредупреждение(, "Для применения промокода вернитесь в режим добавления товаров.");
        ОбработатьСледующийШтрихкодИзОчереди();
        Возврат;
    КонецЕсли;

    Если ПустаяСтрока(СокрЛП(Сред(КодQR, 7))) Тогда
        ПоказатьПредупреждение(, "QR-код не содержит промокод.");
    Иначе
        Расш1_ПромокодПослеВвода(КодQR, Неопределено);
    КонецЕсли;

    ОбработатьСледующийШтрихкодИзОчереди();

КонецПроцедуры

// Освобождение только после штатного удаления неоплаченного чека.
&НаКлиенте
&Вместо("УдалитьЧекКлиент")
Процедура Расш1_УдалитьЧекСРезервом(ОтменитьРежимПлитки = Ложь)

    Если НЕ ЗначениеЗаполнено(Расш1_КодРезерва)
        Или НЕ ЗначениеЗаполнено(Расш1_ТокенРезерва)
        Или Расш1_НачатаФискализация
        Или ПробитыеЧеки.Количество() > 0
        Или ЗначениеЗаполнено(ЧекККМВОбработке)
        Или ЗначениеЗаполнено(ОтложенныйЧек)
        Или Объект.Оплата.Количество() > 0
        Или Объект.Оплачивается
        Или СуммаОплаченоСБП > 0
        Или СуммаОплаченоЮКасса > 0
        Или НЕ (РежимРаботыСЧеком = РежимыРаботыСЧеком.ВводТоваров) Тогда
        ПродолжитьВызов(ОтменитьРежимПлитки);
        Возврат;
    КонецЕсли;

    КодРезерва = Расш1_КодРезерва;
    ТокенРезерва = Расш1_ТокенРезерва;
    Расш1_УдалениеПромочека = Истина;
    Попытка
        ПродолжитьВызов(ОтменитьРежимПлитки);
    Исключение
        Расш1_УдалениеПромочека = Ложь;
        Сообщить("Резерв промокода требует проверки: " + КодРезерва + "; ID: " + ТокенРезерва);
        ВызватьИсключение;
    КонецПопытки;
    Расш1_УдалениеПромочека = Ложь;

    Если Объект.Товары.Количество() <> 0
        Или ПробитыеЧеки.Количество() > 0
        Или ЗначениеЗаполнено(Расш1_КодРезерва) Тогда
        Сообщить("Удаление чека не подтверждено. Резерв требует проверки: " + КодРезерва + "; ID: " + ТокенРезерва);
        Возврат;
    КонецЕсли;

    Попытка
        ОтветAPI = Расш1_ЗапросПромоAPI("release", КодРезерва, ТокенРезерва);
        Если НЕ ОтветAPI.ok Тогда
            ВызватьИсключение ОтветAPI.message;
        КонецЕсли;
        Если ОтветAPI.status <> "active" Тогда
            ВызватьИсключение "Сервер не подтвердил освобождение.";
        КонецЕсли;
    Исключение
        Сообщить("Чек удален, но освобождение резерва не подтверждено: "
            + КодРезерва + "; ID: " + ТокенРезерва + "; " + ОписаниеОшибки());
    КонецПопытки;

КонецПроцедуры

    Кнопка = Элементы.ПрименитьПромокодNew;
    Если ЗначениеЗаполнено(Расш1_ПромокодВТекущемЧеке) Или Расш1_ОтменаПромокодаОжидаетОтвета Тогда

    КонецЕсли;

&НаКлиенте
Процедура Расш1_ОтменитьСобственныйПромокод()
    Если Расш1_НачатаФискализация
        Или ПробитыеЧеки.Количество() > 0
        Или ЗначениеЗаполнено(ЧекККМВОбработке)
        Или ЗначениеЗаполнено(ОтложенныйЧек)
        Или Объект.Оплата.Количество() > 0
        Или Объект.Оплачивается
        Или СуммаОплаченоСБП > 0
        Или СуммаОплаченоЮКасса > 0
        Или ОжидатьВводМарки
        Или НЕ (РежимРаботыСЧеком = РежимыРаботыСЧеком.ВводТоваров)
        Или НЕ ОбщегоНазначенияРМККлиент.ЭтоВидОперацииПродажа(Объект.ВидОперации)
        Или ЭтоЧекКоррекции Тогда
        ПоказатьПредупреждение(, "Отмена промокода доступна только до оплаты и фискализации.");
        Возврат;
    КонецЕсли;
    Если НЕ ЗначениеЗаполнено(Расш1_КодРезерва) Или НЕ ЗначениеЗаполнено(Расш1_ТокенРезерва) Тогда
        ПоказатьПредупреждение(, "Не найден резерв промокода. Обратитесь к администратору.");
        Возврат;
    КонецЕсли;

    Попытка
        Если НЕ Расш1_ОтменаПромокодаОжидаетОтвета Тогда
            Если ЗначениеЗаполнено(Расш1_ИдентификаторПодарка) Тогда
                СтрокаПодарка = Неопределено;
                Для Каждого СтрокаТовара Из Объект.Товары Цикл
                    Если Строка(СтрокаТовара.ИдентификаторСтроки) = Расш1_ИдентификаторПодарка Тогда
                        Если НЕ СтрокаТовара.Подарок Тогда
                            ВызватьИсключение "Строка промоподарка изменена.";
                        КонецЕсли;
                        СтрокаПодарка = СтрокаТовара;
                        Прервать;
                    КонецЕсли;
                КонецЦикла;
                Если СтрокаПодарка <> Неопределено Тогда
                    Если ЗначениеЗаполнено(СтрокаПодарка.КодМаркировки) Тогда
                        ВызватьИсключение "Для маркированного подарка используйте отмену всего чека.";
                    КонецЕсли;
                    УдалитьСтрокуТовара(ЭтотОбъект, СтрокаПодарка, Объект.Товары);
                    Для Каждого СтрокаТовара Из Объект.Товары Цикл
                        Если Строка(СтрокаТовара.ИдентификаторСтроки) = Расш1_ИдентификаторПодарка Тогда
                            ВызватьИсключение "Удаление подарка не подтверждено.";
                        КонецЕсли;
                    КонецЦикла;
                    ОбновитьИнформациюНаДисплееПокупателя(ТипыОперацийДисплеяПокупателя.УдалениеТовара);
                КонецЕсли;
            Иначе
                // Штатный расчет ручной процентной скидки с нулевым значением.
                // При применении промокода существующая ручная скидка запрещена.
                РассчитатьРучнуюСкидкуПроцентомНаПозиции(Объект.Товары, 0);
                Для Каждого СтрокаТовара Из Объект.Товары Цикл
                    Если НЕ СтрокаТовара.Подарок И (СтрокаТовара.СуммаСкидкиНаценки <> 0
                        Или СтрокаТовара.ПроцентСкидкиНаценки <> 0) Тогда
                        ВызватьИсключение "Ручная скидка не обнулена. Освобождение резерва не выполнено.";
                    КонецЕсли;
                КонецЦикла;
            КонецЕсли;

            // Сначала убрать льготу локально; только затем разрешить использование на другой кассе.
            Расш1_ПромокодВТекущемЧеке = "";
            Расш1_ИдентификаторПодарка = "";
            Расш1_ОтменаПромокодаОжидаетОтвета = Истина;
            ПересчитатьИтогиТабличнойЧастиТовары(Ложь);
            РассчитатьСуммыОплаты();
            ТребуетсяПересчетСкидок = Истина;
            ОбработатьИзменениеТаблицыТоваровНаКлиенте();
        КонецЕсли;

        ОтветAPI = Расш1_ЗапросПромоAPI("release", Расш1_КодРезерва, Расш1_ТокенРезерва);
        Если НЕ ОтветAPI.ok Тогда
            ВызватьИсключение ОтветAPI.message;
        КонецЕсли;
        Если ОтветAPI.status <> "active" Тогда
            ВызватьИсключение "Сервер не подтвердил освобождение.";
        КонецЕсли;
        Расш1_КодРезерва = "";
        Расш1_ТокенРезерва = "";
        Расш1_ОтменаПромокодаОжидаетОтвета = Ложь;
        Расш1_ОбновитьКнопкуПромокода();
    Исключение
        Расш1_ОбновитьКнопкуПромокода();
        ПоказатьПредупреждение(, "Отмена промокода не подтверждена. Повторите отмену или обратитесь к администратору. "
            + ОписаниеОшибки());
    КонецПопытки;
КонецПроцедуры
