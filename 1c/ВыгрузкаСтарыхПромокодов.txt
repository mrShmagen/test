&НаСервереБезКонтекста
Функция Расш1_ВыгрузитьСтарыеПромокодыНаСервере()
    Запрос = Новый Запрос;
    Запрос.Текст =
    "ВЫБРАТЬ
    |   П.Промокод КАК Промокод,
    |   П.ДатаНачала КАК ДатаНачала,
    |   П.ДатаОкончания КАК ДатаОкончания,
    |   П.ТипПромокода КАК ТипПромокода,
    |   П.Номенклатура.Код КАК SKU,
    |   П.Количество КАК Количество,
    |   П.ПроцентСкидки КАК ПроцентСкидки
    |ИЗ Справочник.Промокоды КАК П
    |ГДЕ НЕ П.ПометкаУдаления И П.Статус = &Статус
    |УПОРЯДОЧИТЬ ПО П.Промокод";
    Запрос.УстановитьПараметр("Статус", Перечисления.СтатусыПромокодов.Активно);
    Выборка = Запрос.Выполнить().Выбрать();
    Пакет = Новый Массив;
    Итоги = Новый Структура("updated,unchanged,missing,protected,conflict", 0, 0, 0, 0, 0);
    Замечания = Новый Массив;
    Пропущено = 0;
    Пока Выборка.Следующий() Цикл
        Если НЕ ЗначениеЗаполнено(Выборка.ДатаНачала)
            Или НЕ ЗначениеЗаполнено(Выборка.ДатаОкончания)
            Или Выборка.ДатаНачала > Выборка.ДатаОкончания Тогда
            Пропущено = Пропущено + 1;
            Продолжить;
        КонецЕсли;
        ПараметрыПромоAPI = Новый Структура;
        ПараметрыПромоAPI.Вставить("valid_from", Формат(Выборка.ДатаНачала, "ДФ=yyyy-MM-dd"));
        ПараметрыПромоAPI.Вставить("valid_until", Формат(Выборка.ДатаОкончания, "ДФ=yyyy-MM-dd"));
        Если Выборка.ТипПромокода = Перечисления.ТипыПромокодов.Подарок Тогда
            Если НЕ ЗначениеЗаполнено(Выборка.SKU) Или Выборка.Количество <= 0 Тогда
                Пропущено = Пропущено + 1;
                Продолжить;
            КонецЕсли;
            ПараметрыПромоAPI.Вставить("promo_type", "gift");
            ПараметрыПромоAPI.Вставить("product_sku", СокрЛП(Выборка.SKU));
            ПараметрыПромоAPI.Вставить("gift_quantity", Выборка.Количество);
        ИначеЕсли Выборка.ТипПромокода = Перечисления.ТипыПромокодов.Скидка Тогда
            Если Выборка.ПроцентСкидки <= 0 Или Выборка.ПроцентСкидки > 100 Тогда
                Пропущено = Пропущено + 1;
                Продолжить;
            КонецЕсли;
            ПараметрыПромоAPI.Вставить("promo_type", "discount");
            ПараметрыПромоAPI.Вставить("discount_percent", Выборка.ПроцентСкидки);
        Иначе
            Пропущено = Пропущено + 1;
            Продолжить;
        КонецЕсли;
        Пакет.Добавить(Новый Структура("code,metadata", ВРег(СокрЛП(Выборка.Промокод)), ПараметрыПромоAPI));
        Если Пакет.Количество() = 100 Тогда
            Расш1_ОтправитьПакетСтарыхПромокодов(Пакет, Итоги, Замечания);
            Пакет.Очистить();
        КонецЕсли;
    КонецЦикла;
    Если Пакет.Количество() > 0 Тогда
        Расш1_ОтправитьПакетСтарыхПромокодов(Пакет, Итоги, Замечания);
    КонецЕсли;
    Возврат "Заполнено: " + Строка(Итоги.updated)
        + "; Уже совпадают: " + Строка(Итоги.unchanged)
        + "; Отсутствуют в MySQL: " + Строка(Итоги.missing)
        + "; Защищены статусом: " + Строка(Итоги.protected)
        + "; Различаются: " + Строка(Итоги.conflict)
        + "; Некорректны в 1С: " + Строка(Пропущено)
        + Символы.ПС + СтрСоединить(Замечания, Символы.ПС);
КонецФункции

Процедура Расш1_ОтправитьПакетСтарыхПромокодов(Пакет, Итоги, Замечания)
    КлючAPI = "REPLACE_WITH_RANDOM_SECRET_KEY";
    Данные = Новый Структура("action,items", "backfill", Пакет);
    Запись = Новый ЗаписьJSON;
    Запись.УстановитьСтроку();
    ЗаписатьJSON(Запись, Данные);
    Тело = Запись.Закрыть();
    Соединение = Новый HTTPСоединение("adm.pivland.ru", 443, , , , 30,
        Новый ЗащищенноеСоединениеOpenSSL);
    Запрос = Новый HTTPЗапрос("/1c/promo_api.php");
    Запрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
    Запрос.Заголовки.Вставить("X-API-Key", КлючAPI);
    Запрос.УстановитьТелоИзСтроки(Тело, КодировкаТекста.UTF8,
        ИспользованиеByteOrderMark.НеИспользовать);
    Ответ = Соединение.ОтправитьДляОбработки(Запрос);
    Чтение = Новый ЧтениеJSON;
    Чтение.УстановитьСтроку(Ответ.ПолучитьТелоКакСтроку());
    Результат = ПрочитатьJSON(Чтение);
    Чтение.Закрыть();
    Если НЕ Результат.ok Тогда
        ВызватьИсключение Результат.message;
    КонецЕсли;
    Если Ответ.КодСостояния <> 200 Тогда
        ВызватьИсключение "Неожиданный HTTP-статус.";
    КонецЕсли;
    Для Каждого Счетчик Из Итоги Цикл
        Итоги[Счетчик.Ключ] = Счетчик.Значение + Результат[Счетчик.Ключ];
    КонецЦикла;
    Для Каждого Замечание Из Результат.issues Цикл
        Замечания.Добавить(Замечание.code + ": " + Замечание.reason);
    КонецЦикла;
