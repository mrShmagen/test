Процедура ВыполнитьВыгрузкуНаСервере(Лимит)
    // Лимит 0 означает все карты. Повторный запуск обновляет записи по GUID.
    Отправлено = 0;
    Пакет = Новый Массив;
    Попытка
        Запрос = Новый Запрос;
        Запрос.Текст =
        "ВЫБРАТЬ
        |   Карты.Ссылка КАК Карта,
        |   Карты.ВладелецКарты КАК Владелец,
        |   Карты.КодКарты КАК КодКарты,
        |   Карты.ВидКарты КАК ВидКарты,
        |   Карты.БонуснаяПрограммаЛояльности КАК Программа,
        |   Карты.ПометкаУдаления КАК ПометкаУдаления,
        |   ФИО.Фамилия КАК Фамилия,
        |   ФИО.Имя КАК Имя,
        |   ФИО.Отчество КАК Отчество,
        |   ВЫРАЗИТЬ(Карты.ВладелецКарты КАК Справочник.ФизическиеЛица).ДатаРождения КАК ДатаРождения
        |ИЗ
        |   Справочник.ИнформационныеКарты КАК Карты
        |   ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.ФИОФизЛиц.СрезПоследних КАК ФИО
        |   ПО Карты.ВладелецКарты = ФИО.ФизЛицо
        |ГДЕ
        |   НЕ Карты.ЭтоГруппа
        |   И Карты.ТипКарты = ЗНАЧЕНИЕ(Перечисление.ТипыИнформационныхКарт.Дисконтная)
        |УПОРЯДОЧИТЬ ПО Карты.Ссылка";
        Если Лимит > 0 Тогда
            Запрос.Текст = СтрЗаменить(Запрос.Текст, "ВЫБРАТЬ", "ВЫБРАТЬ ПЕРВЫЕ " + Формат(Лимит, "ЧГ=0"));
        КонецЕсли;
        Выборка = Запрос.Выполнить().Выбрать();
        Пока Выборка.Следующий() Цикл
            Пакет.Добавить(ДанныеКартыДляСайта(Выборка));
            Если Пакет.Количество() >= 200 Тогда
                ОтправитьПакетКарт(Пакет);
                Отправлено = Отправлено + Пакет.Количество();
                Сообщить("Подтверждено сервером карт: " + Строка(Отправлено));
                Пакет = Новый Массив;
            КонецЕсли;
        КонецЦикла;
        Если Пакет.Количество() > 0 Тогда
            ОтправитьПакетКарт(Пакет);
            Отправлено = Отправлено + Пакет.Количество();
        КонецЕсли;
        Сообщить("Выгрузка завершена. Подтверждено карт: " + Строка(Отправлено));
    Исключение
        Сообщить("Выгрузка остановлена. Подтверждено карт: " + Строка(Отправлено)
            + ". Ошибка: " + ОписаниеОшибки());
    КонецПопытки;
КонецПроцедуры

&НаСервереБезКонтекста
Функция ДанныеКартыДляСайта(Выборка)
    Карта = Выборка.Карта;
    Владелец = Выборка.Владелец;
    Данные = Новый Структура;
    Данные.Вставить("card_guid", Строка(Карта.УникальныйИдентификатор()));
    GUIDВладельца = Неопределено;
    Если ЗначениеЗаполнено(Владелец) Тогда
        GUIDВладельца = Строка(Владелец.УникальныйИдентификатор());
    КонецЕсли;
    Данные.Вставить("owner_guid", GUIDВладельца);
    Штрихкод = "";
    Если Выборка.ВидКарты <> Перечисления.ВидыИнформационныхКарт.Магнитная Тогда
        ДанныеШтрихкода = ПодключаемоеОборудованиеРТ.ШтрихкодТовара(Карта);
        Если ДанныеШтрихкода <> Неопределено Тогда
            Штрихкод = СокрЛП(Строка(ДанныеШтрихкода.Штрихкод));
        КонецЕсли;
    КонецЕсли;
    НомерКарты = СокрЛП(Строка(Выборка.КодКарты));
    Если ПустаяСтрока(НомерКарты) Тогда
        НомерКарты = Штрихкод;
    КонецЕсли;
    Данные.Вставить("card_number", НомерКарты);
    Данные.Вставить("barcode", Штрихкод);
    Данные.Вставить("last_name", ТекстИлиПусто(Выборка.Фамилия));
    Данные.Вставить("first_name", ТекстИлиПусто(Выборка.Имя));
    Данные.Вставить("middle_name", ТекстИлиПусто(Выборка.Отчество));
    Телефон = "";
    Для Каждого СтрокаКИ Из Карта.КонтактнаяИнформация Цикл
        Если СтрокаКИ.Тип = Перечисления.ТипыКонтактнойИнформации.Телефон Тогда
            Телефон = НормализоватьТелефон(СтрокаКИ.НомерТелефона);
            Если Не ПустаяСтрока(Телефон) Тогда
                Прервать;
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;
    Данные.Вставить("phone", Телефон);
    ДатаРождения = Неопределено;
    Если ТипЗнч(Выборка.ДатаРождения) = Тип("Дата") Тогда
        Если ЗначениеЗаполнено(Выборка.ДатаРождения) И Год(Выборка.ДатаРождения) >= 1000 Тогда
            ДатаРождения = Формат(Выборка.ДатаРождения, "ДФ=yyyy-MM-dd");
        КонецЕсли;
    КонецЕсли;
    Данные.Вставить("birth_date", ДатаРождения);
    Программа = "";
    Если ЗначениеЗаполнено(Выборка.Программа) Тогда
        Программа = СокрЛП(Выборка.Программа.Наименование);
    КонецЕсли;
    Данные.Вставить("loyalty_program", Программа);
    Данные.Вставить("cashback_percent", ПроцентКэшбэка(Программа));
    // Баланс пока не выгружаем: будет отдельный этап с дисконтным сервером.
    Данные.Вставить("bonus_balance", Неопределено);
    // Только карты с известным владельцем-физлицом и штрихкодом готовы к сайту.
    Данные.Вставить("is_active", Не Выборка.ПометкаУдаления
        И ТипЗнч(Владелец) = Тип("СправочникСсылка.ФизическиеЛица")
        И ЗначениеЗаполнено(Владелец) И Не ПустаяСтрока(Штрихкод));
    Возврат Данные;
КонецФункции

&НаСервереБезКонтекста
Функция ТекстИлиПусто(Значение)
    Если ЗначениеЗаполнено(Значение) Тогда
        Возврат СокрЛП(Строка(Значение));
    КонецЕсли;
    Возврат "";
КонецФункции

&НаСервереБезКонтекста
Функция НормализоватьТелефон(Телефон)
    Цифры = "";
    Для Индекс = 1 По СтрДлина(Телефон) Цикл
        Символ = Сред(Телефон, Индекс, 1);
        Если Найти("0123456789", Символ) > 0 Тогда
            Цифры = Цифры + Символ;
        КонецЕсли;
    КонецЦикла;
    Если СтрДлина(Цифры) = 10 Тогда
        Возврат "+7" + Цифры;
    ИначеЕсли СтрДлина(Цифры) = 11 И (Лев(Цифры, 1) = "8" Или Лев(Цифры, 1) = "7") Тогда
        Возврат "+7" + Прав(Цифры, 10);
    КонецЕсли;
    Возврат "";
КонецФункции

&НаСервереБезКонтекста
Функция ПроцентКэшбэка(Программа)
    Если Программа = "Кэшбэк 5%" Тогда
        Возврат 5;
    ИначеЕсли Программа = "Кэшбэк 7%" Тогда
        Возврат 7;
    ИначеЕсли Программа = "Кэшбэк 10%" Тогда
        Возврат 10;
    КонецЕсли;
    Возврат Неопределено;
КонецФункции

Процедура ОтправитьПакетКарт(Пакет)
    КлючAPI = "REPLACE_WITH_RANDOM_SECRET_KEY";
    Если КлючAPI = "REPLACE_WITH_RANDOM_SECRET_KEY" Тогда
        ВызватьИсключение "Укажите ключ импорта карт в локальном модуле.";
    КонецЕсли;
    Данные = Новый Структура("action,cards", "upsert_cards", Пакет);
    Запись = Новый ЗаписьJSON;
    Запись.УстановитьСтроку();
    ЗаписатьJSON(Запись, Данные);
    Тело = Запись.Закрыть();
    Соединение = Новый HTTPСоединение("adm.pivland.ru", 443, , , , 30, Новый ЗащищенноеСоединениеOpenSSL);
    HTTPЗапрос = Новый HTTPЗапрос("/1c/loyalty_api.php");
    HTTPЗапрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
    HTTPЗапрос.Заголовки.Вставить("X-API-Key", КлючAPI);
    HTTPЗапрос.УстановитьТелоИзСтроки(Тело, КодировкаТекста.UTF8, ИспользованиеByteOrderMark.НеИспользовать);
    Ответ = Соединение.ОтправитьДляОбработки(HTTPЗапрос);
    Чтение = Новый ЧтениеJSON;
    Чтение.УстановитьСтроку(Ответ.ПолучитьТелоКакСтроку());
    Результат = ПрочитатьJSON(Чтение);
    Чтение.Закрыть();
    Если Ответ.КодСостояния <> 200 Тогда
        ВызватьИсключение "API HTTP " + Строка(Ответ.КодСостояния) + ": " + Ответ.ПолучитьТелоКакСтроку();
    КонецЕсли;
    Если Не Результат.ok Или Результат.count <> Пакет.Количество() Тогда
        ВызватьИсключение "API не подтвердил полный пакет карт.";
    КонецЕсли;
КонецПроцедуры
