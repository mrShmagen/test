&НаКлиенте
Процедура СоздатьПятьКарт(Команда)
    Попытка
        Сообщить(СоздатьРезервНаСервере(5));
    Исключение
        Сообщить("Резерв не создан: " + ОписаниеОшибки());
    КонецПопытки;
КонецПроцедуры

&НаСервереБезКонтекста
Функция СоздатьРезервНаСервере(КоличествоКарт)
    Если КоличествоКарт < 1 Или КоличествоКарт > 200 Тогда
        ВызватьИсключение "Допустимо от 1 до 200 карт.";
    КонецЕсли;
    Префикс = "78236527777";
    Шаблон = Префикс + "_____";
    Программа = Справочники.БонусныеПрограммыЛояльности.НайтиПоНаименованию("Кэшбэк 5%", Истина);
    Если Не ЗначениеЗаполнено(Программа) Тогда
        ВызватьИсключение "Не найдена бонусная программа Кэшбэк 5%.";
    КонецЕсли;
    НачатьТранзакцию();
    Попытка
        // Резерв создается одним оператором в центре. Блокировки
        // защищают этот пакет от параллельного запуска обработки.
        Блокировка = Новый БлокировкаДанных;
        Блокировка.Добавить("Справочник.ИнформационныеКарты").Режим = РежимБлокировкиДанных.Исключительный;
        Блокировка.Добавить("РегистрСведений.Штрихкоды").Режим = РежимБлокировкиДанных.Исключительный;
        Блокировка.Заблокировать();
        // Тип штрихкода и вид дисконтной карты берем из существующей карты,
        // а не подставляем EAN13 для 16-значного номера.
        Запрос = Новый Запрос;
        Запрос.Текст =
        "ВЫБРАТЬ ПЕРВЫЕ 1
        |   ШК.ТипШтрихкода КАК ТипШтрихкода,
        |   Карты.ВидДисконтнойКарты КАК ВидДисконтнойКарты
        |ИЗ Справочник.ИнформационныеКарты КАК Карты
        |   ВНУТРЕННЕЕ СОЕДИНЕНИЕ РегистрСведений.Штрихкоды КАК ШК
        |   ПО ШК.Владелец = Карты.Ссылка
        |ГДЕ НЕ Карты.ПометкаУдаления
        |   И НЕ Карты.ЭтоГруппа
        |   И Карты.ТипКарты = ЗНАЧЕНИЕ(Перечисление.ТипыИнформационныхКарт.Дисконтная)
        |   И Карты.БонуснаяПрограммаЛояльности = &Программа
        |   И ШК.Штрихкод ПОДОБНО &Шаблон";
        Запрос.УстановитьПараметр("Программа", Программа);
        Запрос.УстановитьПараметр("Шаблон", Шаблон);
        Образец = Запрос.Выполнить().Выбрать();
        Если Не Образец.Следующий() Тогда
            ВызватьИсключение "Нет существующей карты Кэшбэк 5% с подходящим 16-значным штрихкодом.";
        КонецЕсли;
        ТипШтрихкода = Образец.ТипШтрихкода;
        ВидДисконтнойКарты = Образец.ВидДисконтнойКарты;
        Если Не ЗначениеЗаполнено(ТипШтрихкода) Тогда
            ВызватьИсключение "У карты-образца не указан тип штрихкода.";
        КонецЕсли;
        Запрос.Текст =
        "ВЫБРАТЬ ШК.Штрихкод КАК Номер
        |ИЗ РегистрСведений.Штрихкоды КАК ШК
        |ГДЕ ШК.Штрихкод ПОДОБНО &Шаблон
        |ОБЪЕДИНИТЬ ВСЕ
        |ВЫБРАТЬ Карты.КодКарты
        |ИЗ Справочник.ИнформационныеКарты КАК Карты
        |ГДЕ Карты.КодКарты ПОДОБНО &Шаблон
        |ОБЪЕДИНИТЬ ВСЕ
        |ВЫБРАТЬ Карты.Наименование
        |ИЗ Справочник.ИнформационныеКарты КАК Карты
        |ГДЕ Карты.Наименование ПОДОБНО &Шаблон";
        Запрос.УстановитьПараметр("Шаблон", Шаблон);
        Выборка = Запрос.Выполнить().Выбрать();
        МаксимальныйНомер = 0;
        Пока Выборка.Следующий() Цикл
            Номер = СокрЛП(Строка(Выборка.Номер));
            Если СтрДлина(Номер) = 16 И Лев(Номер, 11) = Префикс Тогда
                Суффикс = Прав(Номер, 5);
                ТолькоЦифры = Истина;
                Для Индекс = 1 По 5 Цикл
                    Если Найти("0123456789", Сред(Суффикс, Индекс, 1)) = 0 Тогда
                        ТолькоЦифры = Ложь;
                        Прервать;
                    КонецЕсли;
                КонецЦикла;
                Если ТолькоЦифры Тогда
                    МаксимальныйНомер = Макс(МаксимальныйНомер, Число(Суффикс));
                КонецЕсли;
            КонецЕсли;
        КонецЦикла;
        Если МаксимальныйНомер + КоличествоКарт > 99999 Тогда
            ВызватьИсключение "Диапазон номеров исчерпан.";
        КонецЕсли;
        Группа = Справочники.ИнформационныеКарты.НайтиПоНаименованию("Резерв сайта", Истина);
        Если Не ЗначениеЗаполнено(Группа) Тогда
            ГруппаОбъект = Справочники.ИнформационныеКарты.СоздатьГруппу();
            ГруппаОбъект.Наименование = "Резерв сайта";
            ГруппаОбъект.Записать();
            Группа = ГруппаОбъект.Ссылка;
        ИначеЕсли Не Группа.ЭтоГруппа Или Группа.ПометкаУдаления Тогда
            ВызватьИсключение "Резерв сайта должен быть действующей группой справочника.";
        КонецЕсли;
        Номера = Новый Массив;
        Для Индекс = 1 По КоличествоКарт Цикл
            Номер = Префикс + Прав("00000" + Формат(МаксимальныйНомер + Индекс, "ЧГ=0;ЧДЦ=0"), 5);
            Карта = Справочники.ИнформационныеКарты.СоздатьЭлемент();
            Карта.Родитель = Группа;
            Карта.Наименование = Номер;
            Карта.ТипКарты = Перечисления.ТипыИнформационныхКарт.Дисконтная;
            Карта.ВидКарты = Перечисления.ВидыИнформационныхКарт.Штриховая;
            Карта.ДатаОткрытия = ТекущаяДатаСеанса();
            Карта.БонуснаяПрограммаЛояльности = Программа;
            Карта.ВидДисконтнойКарты = ВидДисконтнойКарты;
            Карта.Записать();
            Набор = РегистрыСведений.Штрихкоды.СоздатьНаборЗаписей();
            Набор.Отбор.Владелец.Установить(Карта.Ссылка);
            Запись = Набор.Добавить();
            Запись.Владелец = Карта.Ссылка;
            Запись.ТипШтрихкода = ТипШтрихкода;
            Запись.Штрихкод = Номер;
            Если Не Набор.ПроверитьЗаполнение() Тогда
                ВызватьИсключение "Не заполнены обязательные поля штрихкода.";
            КонецЕсли;
            Набор.Записать();
            Номера.Добавить(Номер);
        КонецЦикла;
        ЗафиксироватьТранзакцию();
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;
    Возврат "Создано карт: " + Строка(КоличествоКарт) + Символы.ПС + СтрСоединить(Номера, Символы.ПС);
КонецФункции
