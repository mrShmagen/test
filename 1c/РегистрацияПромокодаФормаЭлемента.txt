Процедура Расш1_ПослеЗаписиПромокода(ПараметрыЗаписи)
    Попытка
        Ответ = Расш1_ЗарегистрироватьПромокодНаСервере(Объект.Ссылка);
        Если НЕ Ответ.ok Тогда
            ПоказатьПредупреждение(, "Промокод сохранен в 1С, но регистрация на сервере не подтверждена: "
                + Ответ.message + ". Для повторной попытки сохраните элемент еще раз.");
        ИначеЕсли Ответ.status = "used" Тогда
            ПоказатьПредупреждение(, "В MySQL этот промокод уже использован. Повторная активация запрещена.");
        ИначеЕсли Ответ.status = "disabled" Тогда
            ПоказатьПредупреждение(, "В MySQL этот промокод отключен. Повторная активация не выполнена.");
        ИначеЕсли Ответ.status = "reserved" Тогда
            ПоказатьПредупреждение(, "В MySQL этот промокод уже зарезервирован. Резерв сохранен.");
        КонецЕсли;
    Исключение
        ПоказатьПредупреждение(, "Промокод сохранен в 1С, но регистрация на сервере не подтверждена. "
            + "Для повторной попытки сохраните элемент еще раз. " + ОписаниеОшибки());
    КонецПопытки;
КонецПроцедуры

&НаСервереБезКонтекста
Функция Расш1_ЗарегистрироватьПромокодНаСервере(СсылкаПромокода)
    // Этот этап регистрирует только активные, не помеченные на удаление записи.
    // Не меняет статус уже существующего кода на сервере.
    ПромокодОбъект = СсылкаПромокода.ПолучитьОбъект();
    Если ПромокодОбъект.ПометкаУдаления
        Или ПромокодОбъект.Статус <> Перечисления.СтатусыПромокодов.Активно Тогда
        Возврат Новый Структура("ok,status", Истина, "skipped");
    КонецЕсли;

    Код = ВРег(СокрЛП(ПромокодОбъект.Промокод));
    Если ПустаяСтрока(Код) Или СтрДлина(Код) > 10 Тогда
        Возврат Новый Структура("ok,message", Ложь, "Некорректный код");
    КонецЕсли;

    КлючAPI = "REPLACE_WITH_RANDOM_SECRET_KEY";
    Данные = Новый Структура("action,code,reservation_id", "create", Код, "");
    Данные.Вставить("metadata", Расш1_ПараметрыПромокодаДляAPI(ПромокодОбъект));
    Запись = Новый ЗаписьJSON;
    Запись.УстановитьСтроку();
    ЗаписатьJSON(Запись, Данные);
    Тело = Запись.Закрыть();
    Соединение = Новый HTTPСоединение("adm.pivland.ru", 443, , , , 10,
        Новый ЗащищенноеСоединениеOpenSSL);
    Запрос = Новый HTTPЗапрос("/1c/promo_api.php");
    Запрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
    Запрос.Заголовки.Вставить("X-API-Key", КлючAPI);
    Запрос.УстановитьТелоИзСтроки(Тело, КодировкаТекста.UTF8,
        ИспользованиеByteOrderMark.НеИспользовать);
    Ответ = Соединение.ОтправитьДляОбработки(Запрос);
    Чтение = Новый ЧтениеJSON;
    Чтение.УстановитьСтроку(Ответ.ПолучитьТелоКакСтроку());
    Результат = ПрочитатьJSON(Чтение);
    Чтение.Закрыть();
    Если Ответ.КодСостояния <> 200 И Результат.ok Тогда
        ВызватьИсключение "Неожиданный HTTP-статус: " + Строка(Ответ.КодСостояния);
    КонецЕсли;
    Если Результат.ok И Результат.status <> "active"
        И Результат.status <> "reserved" И Результат.status <> "used"
        И Результат.status <> "disabled" Тогда
        ВызватьИсключение "Неизвестный статус промокода.";
    КонецЕсли;
    Возврат Результат;
КонецФункции

&НаСервереБезКонтекста
Функция Расш1_ПараметрыПромокодаДляAPI(ДанныеПромокода)
    Метаданные = Новый Структура;
    Метаданные.Вставить("valid_from", Формат(ДанныеПромокода.ДатаНачала, "ДФ=yyyy-MM-dd"));
    Метаданные.Вставить("valid_until", Формат(ДанныеПромокода.ДатаОкончания, "ДФ=yyyy-MM-dd"));
    Если ДанныеПромокода.ТипПромокода = Перечисления.ТипыПромокодов.Подарок Тогда
        Метаданные.Вставить("promo_type", "gift");
        Метаданные.Вставить("product_sku", СокрЛП(ДанныеПромокода.Номенклатура.Код));
        Метаданные.Вставить("gift_quantity", ДанныеПромокода.Количество);
    ИначеЕсли ДанныеПромокода.ТипПромокода = Перечисления.ТипыПромокодов.Скидка Тогда
        Метаданные.Вставить("promo_type", "discount");
        Метаданные.Вставить("discount_percent", ДанныеПромокода.ПроцентСкидки);
    Иначе
        ВызватьИсключение "Неизвестный тип промокода.";
    КонецЕсли;
    Возврат Метаданные;
КонецФункции
