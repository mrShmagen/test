    КонецЕсли;
    Если Объект.Оплачивается Или Объект.Оплата.Количество() > 0 Тогда

    КонецЕсли;
    Оповещение = Новый ОписаниеОповещения("Расш1_ПодтвердитьПривязкуКарты", ЭтотОбъект, Объект.ДисконтнаяКарта);

&НаКлиенте
Процедура Расш1_ПодтвердитьПривязкуКарты(Результат, Карта) Экспорт
    Если Результат <> КодВозвратаДиалога.Да Тогда
        Возврат;
    КонецЕсли;
    Если Объект.ДисконтнаяКарта <> Карта Или Объект.Оплачивается Или Объект.Оплата.Количество() > 0 Тогда
        ПоказатьПредупреждение(, "Карта или состояние чека изменились. Повторите операцию.");
        Возврат;
    КонецЕсли;
    Попытка
        Код = Расш1_ПолучитьКодПривязкиНаСервере(Карта);
        ПоказатьПредупреждение(, "Код привязки карты: " + Код + Символы.ПС
            + "Действует 15 минут. Передайте только владельцу карты.");
    Исключение
        ПоказатьПредупреждение(, "Не удалось выдать код привязки: " + ОписаниеОшибки());
    КонецПопытки;
КонецПроцедуры

&НаСервереБезКонтекста
Функция Расш1_ПолучитьКодПривязкиНаСервере(Карта, Действие = "issue")
    Если ТипЗнч(Карта) <> Тип("СправочникСсылка.ИнформационныеКарты") Или Не ЗначениеЗаполнено(Карта) Тогда
        ВызватьИсключение "Дисконтная карта не выбрана.";
    КонецЕсли;
    Если Карта.ПометкаУдаления Или Карта.ТипКарты <> Перечисления.ТипыИнформационныхКарт.Дисконтная Тогда
        ВызватьИсключение "Карта недоступна для привязки.";
    КонецЕсли;
    Владелец = Карта.ВладелецКарты;
    Если ТипЗнч(Владелец) <> Тип("СправочникСсылка.ФизическиеЛица") Или Не ЗначениеЗаполнено(Владелец) Тогда
        ВызватьИсключение "У карты нет владельца-физлица.";
    КонецЕсли;
    КлючAPI = "REPLACE_WITH_RANDOM_SECRET_KEY";
    Если КлючAPI = "REPLACE_WITH_RANDOM_SECRET_KEY" Тогда
        ВызватьИсключение "Укажите ключ кассы в локальном модуле.";
    КонецЕсли;
    Данные = Новый Структура("action,card_guid,owner_guid", Действие, Строка(Карта.УникальныйИдентификатор()), Строка(Владелец.УникальныйИдентификатор()));
    Запись = Новый ЗаписьJSON;
    Запись.УстановитьСтроку();
    ЗаписатьJSON(Запись, Данные);
    Тело = Запись.Закрыть();
    Соединение = Новый HTTPСоединение("adm.pivland.ru", 443, , , , 10, Новый ЗащищенноеСоединениеOpenSSL);
    HTTPЗапрос = Новый HTTPЗапрос("/1c/loyalty_link_api.php");
    HTTPЗапрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
    HTTPЗапрос.Заголовки.Вставить("X-API-Key", КлючAPI);
    HTTPЗапрос.УстановитьТелоИзСтроки(Тело, КодировкаТекста.UTF8, ИспользованиеByteOrderMark.НеИспользовать);
    Ответ = Соединение.ОтправитьДляОбработки(HTTPЗапрос);
    Чтение = Новый ЧтениеJSON;
    Чтение.УстановитьСтроку(Ответ.ПолучитьТелоКакСтроку());
    Результат = ПрочитатьJSON(Чтение);
    Чтение.Закрыть();
    Если Ответ.КодСостояния = 200 И Результат.ok Тогда
        Если Действие = "status" Тогда
            Возврат Не Результат.linked;
        КонецЕсли;
        Возврат Результат.code;
    КонецЕсли;
    ОшибкаAPI = "";
    Результат.Свойство("error", ОшибкаAPI);
    Если ОшибкаAPI = "already_linked" Тогда
        ВызватьИсключение "Карта уже привязана к аккаунту.";
    ИначеЕсли ОшибкаAPI = "card_unavailable" Тогда
        ВызватьИсключение "Карта отсутствует на сайте или недоступна. Требуется синхронизация.";
    ИначеЕсли ОшибкаAPI = "owner_mismatch" Тогда
        ВызватьИсключение "Владелец карты не совпадает с центральными данными. Требуется синхронизация.";
    ИначеЕсли ОшибкаAPI = "issue_limit" Тогда
        ВызватьИсключение "Слишком много выдач кода. Повторите позднее.";
    ИначеЕсли ОшибкаAPI = "auth" Тогда
        ВызватьИсключение "Ошибка авторизации кассы.";
    КонецЕсли;
    ВызватьИсключение "API HTTP " + Строка(Ответ.КодСостояния) + ": " + ОшибкаAPI;
КонецФункции
