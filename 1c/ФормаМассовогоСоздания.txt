&НаСервере
Функция Расш1_СоздатьПартиюНаСервере()

    Если КоличествоПромокодов < 1 Или КоличествоПромокодов > 1000 Тогда
        ВызватьИсключение "Укажите от 1 до 1000 промокодов.";
    КонецЕсли;
    Если ДлинаПромокода < 1 Или ДлинаПромокода > 10 Тогда
        ВызватьИсключение "Длина промокода должна быть от 1 до 10 символов.";
    КонецЕсли;
    Если НЕ ЗначениеЗаполнено(ДатаНачала) Или НЕ ЗначениеЗаполнено(ДатаОкончания)
        Или НачалоДня(ДатаОкончания) < НачалоДня(ДатаНачала) Тогда
        ВызватьИсключение "Укажите корректный период действия.";
    КонецЕсли;
    Если ТипПромокода = Перечисления.ТипыПромокодов.Подарок Тогда
        Если НЕ ЗначениеЗаполнено(Номенклатура) Или Количество < 1 Тогда
            ВызватьИсключение "Укажите номенклатуру и количество подарков.";
        КонецЕсли;
    ИначеЕсли ТипПромокода = Перечисления.ТипыПромокодов.Скидка Тогда
        Если ПроцентСкидки <= 0 Или ПроцентСкидки > 100 Тогда
            ВызватьИсключение "Укажите процент скидки больше 0 и не более 100.";
        КонецЕсли;
    Иначе
        ВызватьИсключение "Выберите тип промокода.";
    КонецЕсли;

    Если ИспользоватьЦифры И ИспользоватьБуквы Тогда
        Алфавит = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
    ИначеЕсли ИспользоватьБуквы Тогда
        Алфавит = "ABCDEFGHJKLMNPQRSTUVWXYZ";
    ИначеЕсли ИспользоватьЦифры Тогда
        Алфавит = "0123456789";
    Иначе
        ВызватьИсключение "Выберите хотя бы один набор символов.";
    КонецЕсли;

    Партия = НРег(Строка(Новый УникальныйИдентификатор));
    НачатьТранзакцию();
    Попытка
        Расш1_ЗаблокироватьСправочник();
        Если НЕ ПустаяСтрока(Расш1_НезавершеннаяПартия()) Тогда
            ВызватьИсключение "Есть незавершенная партия. Сначала нажмите Повторить отправку.";
        КонецЕсли;

        Запрос = Новый Запрос;
        Запрос.Текст = "ВЫБРАТЬ Промокод ИЗ Справочник.Промокоды";
        Выборка = Запрос.Выполнить().Выбрать();
        Занятые = Новый Соответствие;
        ЗанятоКомбинаций = 0;
        Пока Выборка.Следующий() Цикл
            Код = ВРег(СокрЛП(Выборка.Промокод));
            Если Занятые.Получить(Код) <> Неопределено Тогда
                Продолжить;
            КонецЕсли;
            Занятые.Вставить(Код, Истина);
            Если СтрДлина(Код) = ДлинаПромокода Тогда
                Подходит = Истина;
                Для Индекс = 1 По ДлинаПромокода Цикл
                    Если СтрНайти(Алфавит, Сред(Код, Индекс, 1)) = 0 Тогда
                        Подходит = Ложь;
                        Прервать;
                    КонецЕсли;
                КонецЦикла;
                Если Подходит Тогда
                    ЗанятоКомбинаций = ЗанятоКомбинаций + 1;
                КонецЕсли;
            КонецЕсли;
        КонецЦикла;

        ВсегоКомбинаций = 1;
        Для Индекс = 1 По ДлинаПромокода Цикл
            ВсегоКомбинаций = ВсегоКомбинаций * СтрДлина(Алфавит);
        КонецЦикла;
        Если ВсегоКомбинаций - ЗанятоКомбинаций < КоличествоПромокодов Тогда
            ВызватьИсключение "Недостаточно свободных комбинаций. Увеличьте длину промокода.";
        КонецЕсли;

        Коды = Новый Массив;
        Генератор = Новый ГенераторСлучайныхЧисел;
        Попыток = 0;
        Пока Коды.Количество() < КоличествоПромокодов Цикл
            Попыток = Попыток + 1;
            Если Попыток > Макс(10000, КоличествоПромокодов * 200) Тогда
                ВызватьИсключение "Слишком много совпадений. Увеличьте длину промокода.";
            КонецЕсли;
            Код = "";
            Для Индекс = 1 По ДлинаПромокода Цикл
                Код = Код + Сред(Алфавит, Генератор.СлучайноеЧисло(1, СтрДлина(Алфавит)), 1);
            КонецЦикла;
            Если Занятые.Получить(Код) = Неопределено Тогда
                Занятые.Вставить(Код, Истина);
                Коды.Добавить(Код);
            КонецЕсли;
        КонецЦикла;

        Для Каждого Код Из Коды Цикл
            Элемент = Справочники.Промокоды.СоздатьЭлемент();
            Элемент.Промокод = Код;
            Элемент.Наименование = Код;
            Элемент.ДатаНачала = НачалоДня(ДатаНачала);
            Элемент.ДатаОкончания = НачалоДня(ДатаОкончания);
            Элемент.ТипПромокода = ТипПромокода;
            Если ТипПромокода = Перечисления.ТипыПромокодов.Подарок Тогда
                Элемент.Номенклатура = Номенклатура;
                Элемент.Количество = Количество;
            Иначе
                Элемент.ПроцентСкидки = ПроцентСкидки;
            КонецЕсли;
            Элемент.Статус = Перечисления.СтатусыПромокодов.ОжидаетСинхронизации;
            Элемент.ИдентификаторПартии = Партия;
            Элемент.РазмерПартии = КоличествоПромокодов;
            Элемент.Записать();
        КонецЦикла;
        ЗафиксироватьТранзакцию();
    Исключение
        ОтменитьТранзакцию();
        ВызватьИсключение;
    КонецПопытки;

    // Внешний запрос только после фиксации локальной партии.
    Возврат Расш1_ОтправитьПартию(Партия);

Функция Расш1_КодыОжидающейПартии(СтрокиПартии)
    Если СтрокиПартии.Количество() = 0 Тогда
        ВызватьИсключение "Партия не найдена.";
    КонецЕсли;
    Коды = Новый Массив;

Функция Расш1_ОтправитьПартию(Партия)
    Попытка
        Коды = Расш1_КодыОжидающейПартии(Расш1_ПолучитьСтрокиПартии(Партия));
        Ответ = Расш1_ЗапросСозданияПартии(Партия, Коды);
        Если НЕ Ответ.ok Тогда
            ВызватьИсключение Ответ.message;
        КонецЕсли;
        Если НРег(Ответ.batch_id) <> Партия Или Ответ.count <> Коды.Количество() Тогда
            ВызватьИсключение "Ответ сервера не соответствует партии.";
        КонецЕсли;

        НачатьТранзакцию();
        Попытка
            Расш1_ЗаблокироватьСправочник();
            СтрокиПартии = Расш1_ПолучитьСтрокиПартии(Партия);
            ТекущиеКоды = Расш1_КодыОжидающейПартии(СтрокиПартии);
            Если ТекущиеКоды.Количество() <> Коды.Количество() Тогда
                ВызватьИсключение "Изменилось количество кодов партии.";
            КонецЕсли;
            Для Индекс = 0 По Коды.Количество() - 1 Цикл
                Если ТекущиеКоды[Индекс] <> Коды[Индекс] Тогда
                    ВызватьИсключение "Изменились коды партии.";
                КонецЕсли;
            КонецЦикла;
            Для Каждого СтрокаПартии Из СтрокиПартии Цикл
                Элемент = СтрокаПартии.Ссылка.ПолучитьОбъект();
                Элемент.Статус = Перечисления.СтатусыПромокодов.Активно;
                Элемент.Записать();
            КонецЦикла;
            ЗафиксироватьТранзакцию();
        Исключение
            ОтменитьТранзакцию();
            ВызватьИсключение;
        КонецПопытки;
        Возврат Новый Структура("Успех,Количество,Партия", Истина, Коды.Количество(), Партия);
    Исключение
        Возврат Новый Структура("Успех,Сообщение", Ложь,
            "Партия " + Партия + " сохранена в 1С, но активация не подтверждена. "
            + "Не создавайте ее повторно. Используйте Повторить отправку. "
            + ОписаниеОшибки());
    КонецПопытки;

Функция Расш1_ЗапросСозданияПартии(Партия, Коды)
    КлючAPI = "REPLACE_WITH_RANDOM_SECRET_KEY";
    Данные = Новый Структура("action,batch_id,codes", "create_batch", Партия, Коды);
    Запись = Новый ЗаписьJSON;
    Запись.УстановитьСтроку();
    ЗаписатьJSON(Запись, Данные);
    Тело = Запись.Закрыть();
    Соединение = Новый HTTPСоединение("adm.pivland.ru", 443, , , , 30,
        Новый ЗащищенноеСоединениеOpenSSL);
    Запрос = Новый HTTPЗапрос("/1c/promo_api.php");
    Запрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
    Запрос.Заголовки.Вставить("X-API-Key", КлючAPI);
    Запрос.УстановитьТелоИзСтроки(Тело, КодировкаТекста.UTF8,
        ИспользованиеByteOrderMark.НеИспользовать);
    Ответ = Соединение.ОтправитьДляОбработки(Запрос);
    Чтение = Новый ЧтениеJSON;
    Чтение.УстановитьСтроку(Ответ.ПолучитьТелоКакСтроку());
    Результат = ПрочитатьJSON(Чтение);
    Чтение.Закрыть();
    Если Ответ.КодСостояния <> 200 И Результат.ok Тогда
        ВызватьИсключение "Неожиданный HTTP-статус: " + Строка(Ответ.КодСостояния);
    КонецЕсли;
    Возврат Результат;
